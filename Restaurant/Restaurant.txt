        // Set Main Category Filter (All / Veg / Non-Veg)
        function setMainCategory(cat) {
            currentMainCategory = cat;
            
            ['all', 'veg', 'non-veg'].forEach(c => {
                const btn = document.getElementById(`btn-cat-${c}`);
                if (c === cat) {
                    btn.className = "px-6 py-3 rounded-full text-sm font-semibold tracking-wide transition-all shadow-sm bg-stone-900 text-luxury-gold";
                } else {
                    btn.className = "px-6 py-3 rounded-full text-sm font-semibold tracking-wide transition-all shadow-sm bg-white/80 text-stone-700 hover:bg-white border border-stone-200";
                }
            });

            applyFilters();
        }

        // Set Sub-Category Filter
        function setSubCategory(sub) {
            currentSubCategory = sub;

            ['all', 'starter', 'main', 'dessert', 'beverage'].forEach(s => {
                const btn = document.getElementById(`sub-${s}`);
                if (s === sub) {
                    btn.className = "px-4 py-2 rounded-xl text-xs font-bold uppercase tracking-wider transition-all bg-amber-700 text-white shadow-sm";
                } else {
                    btn.className = "px-4 py-2 rounded-xl text-xs font-bold uppercase tracking-wider transition-all bg-white/60 text-stone-600 hover:bg-white border border-stone-200";
                }
            });

            applyFilters();
        }

        // Real-Time Live Search Function
        function filterDishes() {
            const queryInput = document.getElementById('search-input');
            const clearBtn = document.getElementById('clear-search-btn');
            
            if (queryInput.value.length > 0) {
                clearBtn.classList.remove('hidden');
            } else {
                clearBtn.classList.add('hidden');
            }

            applyFilters();
        }

        function clearSearchInput() {
            const queryInput = document.getElementById('search-input');
            queryInput.value = '';
            document.getElementById('clear-search-btn').classList.add('hidden');
            applyFilters();
        }

        function applyFilters() {
            const query = document.getElementById('search-input').value.toLowerCase().trim();

            const filtered = menuItems.filter(dish => {
                const matchesMain = currentMainCategory === 'all' || dish.category === currentMainCategory;
                const matchesSub = currentSubCategory === 'all' || dish.subcategory === currentSubCategory;
                
                // Search matching name, description, subcategory, or any tags/ingredients
                const matchesSearch = query === '' || 
                    dish.name.toLowerCase().includes(query) || 
                    dish.description.toLowerCase().includes(query) || 
                    dish.subcategory.toLowerCase().includes(query) ||
                    dish.tags.some(t => t.toLowerCase().includes(query));

                return matchesMain && matchesSub && matchesSearch;
            });

            renderDishes(filtered);
        }

        function resetAllFiltersAndSearch() {
            document.getElementById('search-input').value = '';
            document.getElementById('clear-search-btn').classList.add('hidden');
            setMainCategory('all');
            setSubCategory('all');
        }

        function toggleMobileMenu() {
            const menu = document.getElementById('mobile-menu');
            const icon = document.getElementById('mobile-menu-icon');
            menu.classList.toggle('hidden');
            if (menu.classList.contains('hidden')) {
                icon.className = "fa-solid fa-bars text-xl";
            } else {
                icon.className = "fa-solid fa-xmark text-xl";
            }
        }

        function openBookingModal() {
            document.getElementById('booking-modal').classList.remove('hidden');
        }

        function closeBookingModal() {
            document.getElementById('booking-modal').classList.add('hidden');
        }

        function openBookingModalWithDish(dishName) {
            openBookingModal();
            const notesField = document.getElementById('modal-notes');
            if (notesField.value) {
                notesField.value += `, Interested in tasting: ${dishName}`;
            } else {
                notesField.value = `Interested in tasting: ${dishName}`;
            }
        }

        function handleQuickBooking(e) {
            e.preventDefault();
            const name = document.getElementById('qb-name').value;
            const date = document.getElementById('qb-date').value;
            const time = document.getElementById('qb-time').value;
            const guests = document.getElementById('qb-guests').value;

            showToast(`Thank you, ${name}! Your table for ${guests} on ${date} at ${time} is reserved.`);
            e.target.reset();
        }

        function handleModalBooking(e) {
            e.preventDefault();
            const name = document.getElementById('modal-name').value;
            const date = document.getElementById('modal-date').value;
            const time = document.getElementById('modal-time').value;
            const guests = document.getElementById('modal-guests').value;

            closeBookingModal();
            showToast(`Table confirmed for ${name} (${guests}) on ${date} at ${time}.`);
            e.target.reset();
        }

        function handleNewsletter(e) {
            e.preventDefault();
            showToast("You have successfully subscribed to The Grand Horizon privileges.");
            e.target.reset();
        }

        function showToast(message) {
            const toast = document.getElementById('toast');
            const toastMsg = document.getElementById('toast-msg');
            toastMsg.textContent = message;

            toast.classList.remove('translate-y-32', 'opacity-0');
            toast.classList.add('translate-y-0', 'opacity-100');

            setTimeout(() => {
                toast.classList.remove('translate-y-0', 'opacity-100');
                toast.classList.add('translate-y-32', 'opacity-0');
            }, 4500);
        }
